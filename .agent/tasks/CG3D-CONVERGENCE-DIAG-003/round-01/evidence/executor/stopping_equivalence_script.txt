"""Stopping-behaviour A/B replay (CG3D-CONVERGENCE-DIAG-003).

Runs the SAME scripted fake system through two implementations of
`run_hold` and compares every legacy row field plus the dump_frame /
live_ckpt callback sequences:

  base       = cg3d/protocol.py at 0ca191c7cd6f8970fc8af7e20261f4d7f246fda0
               (the task's pre-created base, unchanged stopping logic);
  candidate  = cg3d/protocol.py at the task commit

The point of the replay is that the additive `termination` /
`numerical_health` records must not perturb the stopping decision,
the window/min-step/slope arithmetic, the umax-cap check, the ordering
of the two checks inside one sampling point, or the resume bookkeeping
(it_start / it_offset / dump_frame / live_ckpt).

Run from the repository root, with the project interpreter:

    python .agent_runtime/published_evidence/stopping_equivalence_script.txt

Exit code 0 => all cases identical.  No Taichi kernel, no GPU, no LBM
step is involved: the fake system supplies only the members run_hold
touches (s.step / set_ladder / measure / pore_cells).
"""
import math
import subprocess
import sys
import types

sys.path.insert(0, '.')
sys.path.insert(0, 'tests')

BASE = '0ca191c7cd6f8970fc8af7e20261f4d7f246fda0'

src = subprocess.run(['git', 'show', BASE + ':cg3d/protocol.py'],
                     capture_output=True, text=True, check=True).stdout
src = src.replace('from .diagnostics import ', 'from cg3d.diagnostics import ')
base = types.ModuleType('protocol_base')
exec(compile(src, '<cg3d/protocol.py@%s>' % BASE[:7], 'exec'), base.__dict__)

import test_termination_diagnostics as t   # noqa: E402  (the fake system)
import cg3d.protocol as cand               # noqa: E402


def same(a, b):
    """Recursive comparison with NaN == NaN (non-finite metrics appear in
    some cases by construction)."""
    if isinstance(a, dict):
        return (isinstance(b, dict) and set(a) == set(b)
                and all(same(a[k], b[k]) for k in a))
    if isinstance(a, (list, tuple)):
        return (isinstance(b, (list, tuple)) and len(a) == len(b)
                and all(same(x, y) for x, y in zip(a, b)))
    if isinstance(a, float) and isinstance(b, float):
        if math.isnan(a) or math.isnan(b):
            return math.isnan(a) and math.isnan(b)
    return a == b


flat, drift, spike, nonfin = (t.sample_flat, t.sample_drift,
                              t.sample_umax_spike, t.sample_nonfinite)


def slow_drift(step, n):
    return dict(flat(step, n), s_nw=0.30 + 1e-4 * step)


def nan_u(step, n):
    return dict(nonfin(step, n), u_rms=float('nan'))


def pc_spike(step, n):
    return dict(flat(step, n), pc_measured=0.03 + 0.01 * step)


def umax_late(step, n):
    return dict(flat(step, n), umax=0.5 if step >= 12 else 1e-3)


CASES = [
    ('flat/sat', {}, flat, {}),
    ('flat/multi', {'qs_mode': 'multi'}, flat, {}),
    ('flat/sat/max8', {'max_steps': 8}, flat, {}),
    ('slow-drift/sat', {}, slow_drift, {}),
    ('fast-drift/multi', {'qs_mode': 'multi', 'max_steps': 40}, drift, {}),
    ('umax-spike@3', {}, spike, {}),
    ('umax-spike@12', {}, umax_late, {}),
    ('umax-cap-tight', {'umax_cap': 1e-4}, flat, {}),
    ('nan-pc/sat', {}, nonfin, {}),
    ('nan-pc-nan-u/multi', {'qs_mode': 'multi'}, nan_u, {}),
    ('pc-drift/multi', {'qs_mode': 'multi'}, pc_spike, {}),
    ('every2-win20', {'every': 2, 'qs_window': 20, 'min_steps': 10}, flat, {}),
    ('every5', {'every': 5, 'max_steps': 45}, flat, {}),
    ('resume-it_start4', {'it_start': 4, 'it_offset': 1000}, flat, {}),
    ('frames+offset', {'dump_every': 2, 'it_offset': 7}, flat, {}),
    ('live-ckpt', {}, flat, {'ckpt_every': 6}),
]

bad = 0
print('base = cg3d/protocol.py @ %s' % BASE)
print('candidate = working tree (task commit)\n')
for label, over, fn, extra in CASES:
    cfg = t.ns(**over)
    vb, vc = {'frames': [], 'ckpts': []}, {'frames': [], 'ckpts': []}
    kw_b, kw_c = {}, {}
    if over.get('dump_every'):
        kw_b['dump_frame'] = vb['frames'].append
        kw_c['dump_frame'] = vc['frames'].append
    if extra.get('ckpt_every'):
        kw_b.update(live_ckpt=vb['ckpts'].append, ckpt_every=extra['ckpt_every'])
        kw_c.update(live_ckpt=vc['ckpts'].append, ckpt_every=extra['ckpt_every'])
    rb = base.run_hold(cfg, t.FakeSystem(fn), 0.06, label, **kw_b)
    rc = cand.run_hold(cfg, t.FakeSystem(fn), 0.06, label, **kw_c)
    diff = sorted(k for k in t.LEGACY_ROW_KEYS if not same(rb[k], rc[k]))
    cb_ok = same(vb['frames'], vc['frames']) and same(vb['ckpts'], vc['ckpts'])
    added = sorted(set(rc) - t.LEGACY_ROW_KEYS)
    added_ok = added == ['numerical_health', 'termination']
    term = rc['termination']
    flag_ok = (term['converged'] == (rc['reason'] == 'quasi-steady')
               and term['step_limit_reached'] == (rc['reason'] == 'max-steps')
               and term['safety_limit_triggered'] == (rc['reason'] == 'umax-cap'))
    ok = (not diff) and cb_ok and added_ok and flag_ok
    bad += 0 if ok else 1
    print('%-22s %-12s steps=%-3d legacy_equal=%-5s callbacks_equal=%-5s '
          'added_keys=%-5s flags_match=%-5s %s'
          % (label, rc['reason'], rc['steps'], not diff, cb_ok, added_ok,
             flag_ok, ('DIFF=' + ','.join(diff)) if diff else ''))

print('\ncases=%d mismatches=%d -> %s'
      % (len(CASES), bad, 'IDENTICAL' if not bad else 'DIFFERENT'))
sys.exit(1 if bad else 0)

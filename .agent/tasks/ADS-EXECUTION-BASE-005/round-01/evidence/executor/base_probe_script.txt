"""Scratch probe: dump the Controller-owned records that carry the execution base.

Builds the same fixture shape as the unit test (control READY commit != execution
base), runs one round with executor-published evidence, and prints the published
process.json and provenance.json verbatim. Nothing outside a temp directory is
touched.
"""

from __future__ import annotations

import importlib.util
import json
import subprocess
import sys
import tempfile
from pathlib import Path

REPO = Path(__file__).resolve().parents[1]
SPEC = importlib.util.spec_from_file_location(
    "ads_controller", REPO / ".agent" / "controller" / "controller.py"
)
assert SPEC and SPEC.loader
controller_module = importlib.util.module_from_spec(SPEC)
sys.modules[SPEC.name] = controller_module
SPEC.loader.exec_module(controller_module)
FAKE_ZCODE = REPO / ".agent" / "controller" / "tests" / "fake_zcode.py"


def git(cwd: Path, *args: str) -> str:
    result = subprocess.run(
        ["git", *args], cwd=cwd, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE
    )
    if result.returncode != 0:
        raise SystemExit(f"git {' '.join(args)} failed:\n{result.stderr}")
    return result.stdout


def main() -> int:
    temp = tempfile.TemporaryDirectory()
    root = Path(temp.name)
    remote = root / "remote.git"
    repo = root / "control"
    worktrees = root / "worktrees"
    git(root, "init", "--bare", str(remote))
    git(root, "clone", str(remote), str(repo))
    git(repo, "switch", "-c", "control")
    git(repo, "config", "user.name", "Probe Planner")
    git(repo, "config", "user.email", "planner@example.invalid")

    (repo / ".gitignore").write_text(".agent_runtime/\n", encoding="utf-8")
    (repo / "AGENTS.md").write_text("# Probe agent rules\n", encoding="utf-8")
    git(repo, "add", ".")
    git(repo, "commit", "-m", "accepted cumulative base")
    execution_base = git(repo, "rev-parse", "HEAD").strip()

    (repo / ".agent" / "tasks" / "ADS-DUMMY-001" / "round-01").mkdir(parents=True)
    (repo / ".agent_runtime").mkdir()
    (repo / "control_only.txt").write_text("control only\n", encoding="utf-8")
    (repo / ".agent" / "tasks" / "ADS-DUMMY-001" / "round-01" / "TASK.md").write_text(
        "# TASK\n\n## Scope\n\n### Allowed\n\n- `dummy.txt`\n\n"
        "### Do not modify\n\n- everything else\n\nDUMMY_PUBLISH_EVIDENCE=1\n",
        encoding="utf-8",
    )
    state = {
        "schema_version": "v1-alpha",
        "task_id": "ADS-DUMMY-001",
        "round": 1,
        "max_rounds": 3,
        "revision": 0,
        "status": "READY_FOR_EXECUTION",
        "zcode_session_id": None,
        "candidate_commit": None,
        "execution_base_commit": execution_base,
        "worker": None,
        "updated_at": None,
        "last_error": None,
    }
    (repo / ".agent" / "state.json").write_text(
        json.dumps(state, indent=2) + "\n", encoding="utf-8"
    )
    git(repo, "add", ".")
    git(repo, "commit", "-m", "queue round 1")
    git(repo, "push", "-u", "origin", "control")
    ready_sha = git(repo, "rev-parse", "HEAD").strip()

    config_path = root / "config.json"
    config_path.write_text(
        json.dumps(
            {
                "repository_path": str(repo),
                "remote": "origin",
                "control_branch": "control",
                "task_branch_prefix": "agent-task/",
                "worktree_root": str(worktrees),
                "worker_id": "probe-worker",
                "zcode": {"command": [sys.executable, str(FAKE_ZCODE)], "timeout_seconds": 30},
            },
            indent=2,
        ),
        encoding="utf-8",
    )
    controller = controller_module.Controller(controller_module.Config.load(config_path))
    outcome = controller.run_once()

    def remote_json(path: str) -> dict[str, object]:
        return json.loads(git(remote, "show", f"control:{path}"))

    prefix = ".agent/tasks/ADS-DUMMY-001/round-01/evidence"
    final_state = remote_json(".agent/state.json")
    print("run_once()               :", outcome)
    print("execution_base_commit    :", execution_base)
    print("control ready_sha        :", ready_sha)
    print("ready != base            :", ready_sha != execution_base)
    print("final state.status       :", final_state["status"])
    print("final state.candidate    :", final_state["candidate_commit"])
    print("final state.base         :", final_state["execution_base_commit"])
    print()
    print("control:" + prefix + "/controller/process.json")
    print(json.dumps(remote_json(f"{prefix}/controller/process.json"), indent=2)[:1200])
    print()
    print("control:" + prefix + "/executor/provenance.json")
    print(json.dumps(remote_json(f"{prefix}/executor/provenance.json"), indent=2)[:1200])
    temp.cleanup()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
